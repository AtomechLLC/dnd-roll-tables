DiceMath (frozen object, block `<script id="dice-math">`):
- DICE: [4, 6, 8, 10, 12]
- avg(x) -> (x + 1) / 2
- assertParams(p) -> throws TypeError unless p.ac, p.atk, p.die, p.dmg are all Number.isInteger
- outcome(d20, atk, ac) -> 'fumble' | 'crit' | 'hit' | 'miss'  (checks d20 === 1, then d20 === 20, then d20 + atk >= ac)
- isHit(d20, atk, ac) -> boolean
- damageFor(d20, dieRolls, p) -> integer >= 0   (THE rule; dieRolls = [k] normal, [k1, k2] crit; damage bonus added once; Math.max(0, sum))
- threshold(p) -> clamp(p.ac - p.atk, 2, 20)
- outcomeTable(p) -> [{ d20, kind, cells: number[die] }] x 20 rows, row 20 cells = damageFor(20, [k, k], p)
- stats(p) -> { rows, hitFaces, pHit, pCrit, e20Num, e20Den, e20, perRound, tableSum, threshold }
- closedFormPerRound(p) -> PROJECT.md closed form (test oracle only, valid for dmg >= -1)
- marginals(p) -> { hitNum, dmgNum, den, reason: null | 'cap' | 'floor' | 'nodmg' }
- sweep(p, maxN = 20) -> { attack: int[21], damage: int[21], den }

- parseControl(key, raw, prev) -> number
- fixed(intScaled, decimals) -> string   (fixed(1575, 3) === "1.575")
